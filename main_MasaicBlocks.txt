import tkinter as tk
from PIL import Image, ImageTk


class MainMosaic:
    def __init__(self):
        self.image = None
        self.photo = None
        self.label = None
        self.image_size = None
        self.original_aspect_ratio = 1.0  # Standardwert

    def load_image(self, path):
        """
        Lädt ein Bild von einem Pfad und speichert es für die weitere Bearbeitung.

        Unterstützte Formate hängen von Pillow ab, z. B.:
        JPG, JPEG, PNG, BMP, GIF, TIFF, WebP
        """
        try:
            self.image = Image.open(path)

            # Bilddaten tatsächlich einlesen und Datei danach schließen können
            self.image.load()

            # Einheitliches Format für spätere Bearbeitung
            if self.image.mode not in ("RGB", "RGBA"):
                self.image = self.image.convert("RGBA")

            print(f"Bild geladen: {path}")
            print(f"Größe: {self.image.size}")
            self.image_size = self.image.size
            print(f"Format: {self.image.format}")

        except FileNotFoundError:
            print(f"Datei nicht gefunden: {path}")

        except Exception as e:
            print(f"Bild konnte nicht geladen werden: {e}")

    def _resize_image(self, event):
        """
        Passt die Bildgröße an die Fenstergröße an, BEHÄLT aber das Seitenverhältnis bei.
        Fügt schwarze Ränder hinzu, um den leeren Platz zu füllen (Letterboxing).
        """
        window_width = event.width
        window_height = event.height

        # Seitenverhältnis des Fensters berechnen
        window_aspect_ratio = window_width / window_height

        # Neue Bildgröße basierend auf dem limitierenden Faktor (Breite oder Höhe) berechnen
        if window_aspect_ratio > self.original_aspect_ratio:
            # Fenster ist breiter als das Bild -> Höhe ist der limitierende Faktor
            new_height = window_height
            new_width = int(new_height * self.original_aspect_ratio)
        else:
            # Fenster ist höher/schmaler als das Bild -> Breite ist der limitierende Faktor
            new_width = window_width
            new_height = int(new_width / self.original_aspect_ratio)

        # Skaliere das Originalbild auf die berechnete, seitenverhältnis-korrekte Größe
        resized_photo = self.image.resize((new_width, new_height), Image.LANCZOS)

        # Erstelle ein neues, leeres Bild mit der vollen Fenstergröße (für die Ränder)
        # Wir verwenden hier einen schwarzen Hintergrund. 'RGBA' ist wichtig für Transparenz.
        final_image = Image.new("RGBA", (window_width, window_height), (0, 0, 0, 255))

        # Berechne die Position, um das skalierte Bild im neuen Hintergrund zu zentrieren
        paste_x = (window_width - new_width) // 2
        paste_y = (window_height - new_height) // 2

        # Füge das skalierte Bild in den zentrierten Bereich ein
        final_image.paste(resized_photo, (paste_x, paste_y))

        # Aktualisiere das PhotoImage und das Label
        self.photo = ImageTk.PhotoImage(final_image)
        self.label.config(image=self.photo)

    def show_image(self, initial_height):
        """Zeigt das Bild in einem Fenster an, das mitskaliert wird."""
        if self.image is None:
            print("Kein Bild geladen.")
            return

        window = tk.Tk()
        window.title("MosaicBlocks")
        # Setzen einer Mindestgröße, um ein zu kleines Fenster zu verhindern
        window.minsize(100, 100)

        # 1. Bild initial auf eine Startgröße skalieren (z.B. 400x400)

        if self.image_size is not None:
            resize_fact = initial_height / self.image_size[1]
            initial_width = int(self.image_size[0] * resize_fact)
        else:
            initial_width = initial_height

        initial_size = (initial_width, initial_height)
        resized_image = self.image.resize(initial_size, Image.LANCZOS)
        self.photo = ImageTk.PhotoImage(resized_image)

        # 2. Label mit dem initial skalierten Bild erstellen
        self.label = tk.Label(window, image=self.photo)
        self.label.pack(fill="both", expand=True)

        # 3. Fenstergeometrie auf die initiale Bildgröße setzen
        window.geometry(f"{initial_size[0]}x{initial_size[1]}")

        # 4. Erst jetzt das Event für zukünftige Größenänderungen binden
        window.bind("<Configure>", self._resize_image)

        window.mainloop()


if __name__ == "__main__":
    Mosaic = MainMosaic()
    Mosaic.load_image("Bilder/Test1.jpg")
    Mosaic.show_image(initial_height=400)
